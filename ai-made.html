function flipMemoryCard(i) { // Prevent clicking if game is paused or card is already flipped if (gameState.memory.flipped[i] || gameState.memory.checking) return; gameState.memory.flipped[i] = true; renderMemory(); const flippedIndexes = gameState.memory.flipped.map((item, index) => (item ? index : -1)).filter(index => index >= 0); if (flippedIndexes.length === 2) { gameState.memory.checking = true; setTimeout(() => { const [a, b] = flippedIndexes; if (gameState.memory.cards[a] === gameState.memory.cards[b]) { // Matched! Keep them flipped gameState.memory.matched += 1; gameState.memory.flipped[a] = 'matched'; gameState.memory.flipped[b] = 'matched'; if (gameState.memory.matched === 8) { document.getElementById('gameScore').textContent = 'You Won! 🎉'; gameState.memory.checking = false; stopGameLoop(); return; } } else { // No match, flip them back gameState.memory.flipped[a] = false; gameState.memory.flipped[b] = false; } gameState.memory.checking = false; renderMemory(); }, 600); } } function renderMemory() { const gameArea = document.getElementById('gameArea'); gameArea.innerHTML = ''; const grid = document.createElement('div'); grid.className = 'memory-grid'; gameState.memory.cards.forEach((card, i) => { const btn = document.createElement('button'); btn.className = 'memory-card'; if (gameState.memory.flipped[i] === 'matched') { btn.classList.add('flipped'); btn.style.opacity = '0.5'; btn.disabled = true; } else if (gameState.memory.flipped[i]) { btn.classList.add('flipped'); } btn.textContent = gameState.memory.flipped[i] ? card : '?'; btn.onclick = () => flipMemoryCard(i); grid.appendChild(btn); }); gameArea.appendChild(grid); document.getElementById('gameScore').textContent = `Matched: ${gameState.memory.matched}/8`; }
